interface PeticionRegistrada {
  metodo: string;
  url: string;
  uri: string;
  config: InternalAxiosRequestConfig;
}

let peticiones: PeticionRegistrada[] = [];
const adaptadorOriginal = http.defaults.adapter;

function adaptadorSimulado(config: InternalAxiosRequestConfig): Promise<AxiosResponse> {
  const metodo = (config.method ?? 'get').toUpperCase();
  const uri = http.getUri(config);
  peticiones.push({ metodo, url: config.url ?? '', uri, config });

  const url = config.url ?? '';

  // a) Opciones de activo
  if (url.includes('/reportes/opciones/activo')) {
    const data = [
      { valor: 885, etiqueta: 'QA-885 · Bovino' },
      { valor: 2945, etiqueta: 'QA-2945 · Bovino' },
    ];
    return Promise.resolve({ data, status: 200, statusText: 'OK', headers: {}, config, request: {} });
  }

  // b) Historial predictivo por id
  const matchHp = url.match(/\/historial-predictivo\/(\d+)/);
  if (matchHp) {
    const idActivo = Number(matchHp[1]);
    const queryStr = uri.includes('?') ? uri.slice(uri.indexOf('?') + 1) : '';
    const params = new URLSearchParams(queryStr);

    const fecha_inicio = params.get('fecha_inicio') ?? '2025-11-20';
    const fecha_fin = params.get('fecha_fin') ?? '2026-11-20';
    const nivelesParam = params.getAll('nivel_riesgo').map(Number);
    const incluirEventos = params.get('incluir_eventos_sanitarios') !== 'false';
    const incluirRetro = params.get('incluir_retroalimentacion') !== 'false';
    const modoVis = (params.get('modo_visualizacion') as 'LINEAL' | 'AGRUPADO') || 'LINEAL';

    // Regla: fecha_inicio > fecha_fin -> 400 VAL_ENTRADA
    if (fecha_inicio > fecha_fin) {
      const body = {
        error_code: 'VAL_ENTRADA',
        message: 'La fecha de inicio no puede ser posterior a la fecha de fin.',
        fields: [{ field: 'fecha_inicio', message: 'La fecha de inicio no puede ser posterior a la fecha de fin.' }],
        timestamp: '2026-11-20T17:00:00Z',
      };
      const respuesta: AxiosResponse = { data: body, status: 400, statusText: 'Bad Request', headers: {}, config, request: {} };
      return Promise.reject(new AxiosError('Request failed with status code 400', 'ERR_BAD_REQUEST', config, {}, respuesta));
    }

    // Regla: fecha_inicio < fecha_fin - 24 meses de calendario -> 400 VAL_ENTRADA
    // (Es el comportamiento del Swagger de TEST, no el defecto INC de G018: déjalo comentado así en el código.)
    const [finY, finM, finD] = fecha_fin.split('-').map(Number);
    const limite24m = new Date(Date.UTC(finY - 2, finM - 1, finD)).toISOString().slice(0, 10);
    if (fecha_inicio < limite24m) {
      const body = {
        error_code: 'VAL_ENTRADA',
        message: 'El rango máximo es de 24 meses. Elige un rango más corto.',
        fields: [{ field: 'fecha_inicio', message: 'El rango máximo es de 24 meses. Elige un rango más corto.' }],
        timestamp: '2026-11-20T17:00:00Z',
      };
      const respuesta: AxiosResponse = { data: body, status: 400, statusText: 'Bad Request', headers: {}, config, request: {} };
      return Promise.reject(new AxiosError('Request failed with status code 400', 'ERR_BAD_REQUEST', config, {}, respuesta));
    }

    // Rango de fechas de Bogotá (05:00:00Z a sigDia 04:59:59.999Z)
    const inicioUtc = new Date(`${fecha_inicio}T05:00:00Z`).getTime();
    const [fY, fM, fD] = fecha_fin.split('-').map(Number);
    const sigDia = new Date(Date.UTC(fY, fM - 1, fD + 1)).toISOString().slice(0, 10);
    const finUtc = new Date(`${sigDia}T04:59:59.999Z`).getTime();

    const enRango = (fechaIso: string) => {
      const t = new Date(fechaIso).getTime();
      return t >= inicioUtc && t <= finUtc;
    };

    const prediccionesEnRango = TODAS_PREDICCIONES.filter((p) => enRango(p.fecha));
    const eventosEnRango = TODOS_EVENTOS.filter((e) => enRango(e.fecha));
    const retrosEnRango = TODAS_RETROS.filter((r) => enRango(r.fecha));

    // nivel_riesgo filtra SOLO las predicciones devueltas; sin nivel_riesgo, todas.
    const prediccionesDevueltas = nivelesParam.length > 0
      ? prediccionesEnRango.filter((p) => nivelesParam.includes(p.nivel_riesgo ?? 0))
      : [...prediccionesEnRango];

    // incluir_eventos_sanitarios=false -> eventos_sanitarios []; incluir_retroalimentacion=false -> retroalimentaciones [].
    const eventosDevueltos = eventosEnRango;
    const retrosDevueltas = incluirRetro ? retrosEnRango : [];

    // metadatos se calcula SIN aplicar nivel ni incluir_*:
    // total_inferencias = predicciones en rango;
    // total_eventos_sanitarios y total_retroalimentaciones = los del rango;
    // total_pendientes = predicciones en rango sin valoración y con más de 48 h;
    // total_conflictos 0.
    const retrosIds = new Set(retrosEnRango.map((r) => r.id_resultado_inferencia));
    const hace48h = new Date('2026-11-18T17:00:00Z').getTime();
    const pendientesPreds = prediccionesEnRango.filter(
      (p) => !retrosIds.has(p.id_resultado_inferencia) && new Date(p.fecha).getTime() <= hace48h,
    );

    const total_inferencias = prediccionesEnRango.length;
    const total_eventos_sanitarios = eventosEnRango.length;
    const total_retroalimentaciones = retrosEnRango.length;
    const total_pendientes = pendientesPreds.length;
    const total_conflictos = 0;

    // tvco:
    // validadas = predicciones en rango con valoración en rango;
    // porcentaje = CORRECTO/validadas*100 (null si 0);
    // texto = `${Math.round(porcentaje)}% (${validadas} validadas)` o "Sin datos suficientes para calcular";
    // tvco_denominador = validadas;
    // tvco_excluidas_conflicto 0;
    // excluidas_sin_retroalimentacion = total_inferencias − validadas;
    // tasa_error = 100 − porcentaje.
    const validadasPreds = prediccionesEnRango.filter((p) => retrosIds.has(p.id_resultado_inferencia));
    const validadas = validadasPreds.length;
    const correctas = retrosEnRango.filter(
      (r) => r.valoracion === 'CORRECTO' && prediccionesEnRango.some((p) => p.id_resultado_inferencia === r.id_resultado_inferencia),
    ).length;

    const porcentaje = validadas > 0 ? (correctas / validadas) * 100 : null;
    const textoTvco = porcentaje !== null ? `${Math.round(porcentaje)}% (${validadas} validadas)` : 'Sin datos suficientes para calcular';
    const tasa_error = porcentaje !== null ? 100 - porcentaje : null;

    const tvco = {
      porcentaje,
      texto: textoTvco,
      tvco_denominador: validadas,
      tvco_excluidas_conflicto: 0,
      excluidas_sin_retroalimentacion: total_inferencias - validadas,
      tasa_error,
    };

    // pendientes: {total, pagina:1, por_pagina:20, items:[{id_resultado_inferencia, fecha_inferencia, nivel_riesgo, condicion}]}
    // con las predicciones de total_pendientes, nivel DESC y fecha DESC.
    const itemsPendientes = [...pendientesPreds]
      .sort((a, b) => (b.nivel_riesgo ?? 0) - (a.nivel_riesgo ?? 0) || new Date(b.fecha).getTime() - new Date(a.fecha).getTime())
      .map((p) => ({
        id_resultado_inferencia: p.id_resultado_inferencia ?? '',
        fecha_inferencia: p.fecha,
        nivel_riesgo: p.nivel_riesgo ?? 0,
        condicion: p.condicion ?? 'Mastitis',
      }));

    const pendientes = {
      total: total_pendientes,
      pagina: 1,
      por_pagina: 20,
      items: itemsPendientes,
    };

    // grupos: pares de marcadores DEVUELTOS (predicciones, eventos y valoraciones) a 60 min o menos,
    // cada par como un arreglo de dos marcadores; con todos los niveles y todo incluido sale solo [p7, 2686].
    const marcadoresDevueltos = [
      ...prediccionesDevueltas,
      ...eventosDevueltos,
      ...retrosDevueltas,
    ].sort((a, b) => new Date(a.fecha).getTime() - new Date(b.fecha).getTime());

    const grupos: MarcadorRaw[][] = [];
    for (let i = 0; i < marcadoresDevueltos.length - 1; i++) {
      const m1 = marcadoresDevueltos[i];
      const m2 = marcadoresDevueltos[i + 1];
      const diffMin = Math.abs(new Date(m2.fecha).getTime() - new Date(m1.fecha).getTime()) / (60 * 1000);
      if (diffMin <= 60) {
        grupos.push([m1, m2]);
      }
    }

    const activo = {
      id_activo_biologico: idActivo,
      identificador: `QA-${idActivo}`,
      especie: 'Bovino',
      finca: 'Finca QA 184',
      estado: idActivo === 2945 ? 'EN_TRATAMIENTO' : 'ACTIVO',
    };

    const dataRespuesta = {
      activo,
      rango_fechas_aplicado: { fecha_inicio, fecha_fin },
      ventana_correspondencia_horas: 12,
      modo_visualizacion: modoVis,
      predicciones: prediccionesDevueltas,
      eventos_sanitarios: eventosDevueltos,
      retroalimentaciones: retrosDevueltas,
      grupos,
      correspondencias: [],
      tvco,
      pendientes,
      metadatos: {
        total_inferencias,
        total_eventos_sanitarios,
        total_retroalimentaciones,
        total_pendientes,
        total_conflictos,
      },
      fuentes_no_disponibles: [],
      mensaje: null,
    };

    const respuesta: AxiosResponse = {
      data: dataRespuesta,
      status: 200,
      statusText: 'OK',
      headers: {},
      config,
      request: {},
    };
    return Promise.resolve(respuesta);
  }

  // c) Cualquier otra petición -> 404
  const body404 = {
    error_code: 'RUTA_NO_SIMULADA',
    message: `Ruta no simulada: ${metodo} ${url}`,
    fields: [],
    timestamp: '2026-11-20T17:00:00Z',
  };
  const respuesta404: AxiosResponse = { data: body404, status: 404, statusText: 'Not Found', headers: {}, config, request: {} };

}

function contarPuntosLinea(): number {
  return document.querySelectorAll('.recharts-line-dots circle').length;
}

function contarMarcadoresEvento(): number {
  return document.querySelectorAll('.recharts-scatter-symbol path').length;
}

function contarMarcadoresValoracion(): number {
  return document.querySelectorAll('.recharts-scatter-symbol text').length;
}

/** Texto de la TVCO tal como se ve (la tarjeta pone el valor justo despues de la etiqueta). */
function leerTvco(): string {
  const etiqueta = screen.queryByText('Tasa de validación clínica');
  return etiqueta?.nextElementSibling?.textContent ?? '(no visible)';
}

/** Texto del total de predicciones tal como se ve. */
function leerTotalInferencias(): string {
  return screen.queryByText(/^\d+ predicci(ón|ones)$/)?.textContent ?? '(no visible)';
}

/** Firma de lo que hay en pantalla; no depende de ningun esperado. */
function firma(): string {
  return [contarPuntosLinea(), contarMarcadoresEvento(), contarMarcadoresValoracion(), leerTvco(),
    screen.queryByText('No se pudo cargar el historial') !== null].join('|');
}

/**
 * Espera a que termine la carga y a que la grafica quede quieta: la misma firma en dos lecturas seguidas
 * separadas 50 ms (recharts puede redibujar despues de que desaparece "Cargando historial...").
 */
async function esperarEstable(): Promise<void> {
  await waitFor(() => expect(screen.queryByText('Cargando historial...')).toBeNull(), { timeout: 3000 });
  let anterior = '';
  for (let i = 0; i < 60; i++) {
    await new Promise((r) => setTimeout(r, 50));
    const actual = firma();
    if (actual === anterior) return;
    anterior = actual;
  }
}

function montar(rol: 'VETERINARIO' | 'ADMINISTRADOR' | 'PRODUCTOR' = 'VETERINARIO', permisos = PERMISOS_VET) {
  const auth = crearAuth(rol, permisos);
  return render(
    <AuthContext.Provider value={auth}>
      <MemoryRouter initialEntries={['/prediccion/historial-predictivo']}>
        <PrediccionPage />
      </MemoryRouter>
    </AuthContext.Provider>,
  );
}

async function seleccionarActivo(etiqueta: string): Promise<void> {
  const combo = await screen.findByRole('combobox', { name: /Animal o lote/ });
  await waitFor(() => expect(peticiones.some((p) => p.url.includes('/reportes/opciones/activo'))).toBe(true));
  await new Promise((r) => setTimeout(r, 0));
  fireEvent.change(combo, { target: { value: etiqueta } });
  await esperarEstable();
}
