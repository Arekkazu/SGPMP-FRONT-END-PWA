"""Controles por mutacion (Vitest, G021 reintento 1). Mutan una COPIA del contenedor en modo binario y la restauran."""
import json, os, subprocess, sys

FE = '/home/claude/g020r1/fe'
VISTA = 'src/prediction/pages/HistorialPredictivoView.tsx'
SPEC = 'testing/test_testing/Modulo8/RF-106/TC-M08-G021/tc-m08-g021.test.tsx'

CONTROLES = [
    # sin conectores: la vista no dibuja ningun ReferenceLine
    ('sin_conectores', VISTA, [(b"const conectores = (datos?.correspondencias ?? []).filter((c) => !c.colision).flatMap((c) => {",
                                b"const conectores = (datos?.correspondencias ?? []).filter(() => false).flatMap((c) => {")],
     ['CP-01', 'CP-02', 'CP-03', 'CP-05', 'CP-08']),
    # conector tambien en las colisiones (no cambia ningun checkpoint de presencia; los pares fuera siguen sin conector)
    ('sin_marcador_grupo', VISTA, [(b"<Scatter data={gruposPts}", b"<Scatter data={[]}")], ['CP-06', 'CP-07']),
    # el simulador amplia la ventana a 13 h: el par de 12 h 01 queda vinculado
    ('ventana_13', SPEC, [(b"const VENTANA_SIMULADOR = 12;", b"const VENTANA_SIMULADOR = 13; // CONTROL ventana_13"),
        (b"4001: ['P1-E1 600 \"Diferencia: 10 h\" false', 'P2-E2 720 \"Diferencia: 12 h\" false', 'P5-E5 600 \"Diferencia: 10 h\" false'],",
         b"4001: ['P1-E1 600 \"Diferencia: 10 h\" false', 'P2-E2 720 \"Diferencia: 12 h\" false', 'P3-E3 721 \"Diferencia: 12 h 1 min\" false', 'P5-E5 600 \"Diferencia: 10 h\" false'],")],
     ['CP-04', 'CP-05']),
    # el simulador desordena el grupo
    ('grupo_desorden', SPEC, [(b"const ORDEN_GRUPO_BACKEND = true;", b"const ORDEN_GRUPO_BACKEND = false; // CONTROL grupo_desorden"),
        (b"const GRUPOS_ESPERADOS: Record<number, string[]> = { 4001: [], 4002: ['E6,P6', 'E7,P7', 'R9,E9,P9'] };",
         b"const GRUPOS_ESPERADOS: Record<number, string[]> = { 4001: [], 4002: ['P6,E6', 'P7,E7', 'P9,E9,R9'] };")], ['CP-09']),
    # evento como path (mecanismo de la ronda 1): los conectores salen del centro del rect; con path en el mismo centro todo OK
    ('evento_path_ronda1', VISTA, [
        (b'<rect x={(p.cx ?? 0) - 5} y={(p.cy ?? 0) - 5} width={10} height={10} fill="var(--sem-error)">',
         b'<path d={`M${(p.cx ?? 0) - 5},${(p.cy ?? 0) - 5}L${(p.cx ?? 0) + 5},${(p.cy ?? 0) + 5}`} fill="var(--sem-error)">'),
        (b'</rect>', b'</path>')], []),
]

salida = []
for nombre, rel, reemplazos, esperadas in CONTROLES:
    ruta = os.path.join(FE, rel)
    original = open(ruta, 'rb').read()
    mut = original
    for viejo, nuevo in reemplazos:
        n = mut.count(viejo)
        assert n == 1, f'{nombre}: {n} apariciones de {viejo[:70]!r}'
        mut = mut.replace(viejo, nuevo)
    st = f'/home/claude/g021r1/vt_ctl_{nombre}'
    try:
        open(ruta, 'wb').write(mut)
        env = dict(os.environ, TZ='America/Bogota', STAGING_DIR=st, REINTENTO='1', RUN_ID=f'control-{nombre}')
        p = subprocess.run(['pnpm', 'exec', 'vitest', 'run', SPEC, '--reporter=verbose'], cwd=FE, env=env, capture_output=True, text=True, timeout=400)
        open(f'{st}_consola.txt', 'w').write(p.stdout + p.stderr)
    finally:
        open(ruta, 'wb').write(original)
    rep = json.load(open(f'{st}/parcial_vitest_TC-M08-G021_reintento1.json'))
    fallas = [c['paso'] for c in rep['checkpoints'] if c['estado'] == 'FALLA']
    no_alc = [c['paso'] for c in rep['checkpoints'] if 'No alcanzado' in c['obtenido']]
    prec = all(x['ok'] for x in rep['precondiciones'])
    ok = sorted(fallas) == sorted(esperadas) and not no_alc
    linea = {'control': nombre, 'exit': p.returncode, 'precondiciones_ok': prec, 'fallas': fallas, 'esperadas': esperadas, 'no_alcanzado': no_alc,
             'como_previsto': ok, 'obtenidos_falla': {c['paso']: c['obtenido'][:240] for c in rep['checkpoints'] if c['estado'] == 'FALLA'}}
    salida.append(linea)
    print(json.dumps(linea, ensure_ascii=False))
json.dump(salida, open('/home/claude/g021r1/controles_vitest_resumen.json', 'w'), ensure_ascii=False, indent=2)
sys.exit(0 if all(x['como_previsto'] for x in salida) else 1)
