Uso: editar_g022.py <carpeta del caso>"""
import sys, os

caso = sys.argv[1]


def aplicar(nombre, cambios, binario=False):
    ruta = os.path.join(caso, nombre)
    s = open(ruta, 'rb').read() if binario else open(ruta, encoding='utf-8', newline='').read()
    for viejo, nuevo in cambios:
        n = s.count(viejo)
        assert n == 1, f'{nombre}: {n} apariciones de {viejo[:100]!r}'
        s = s.replace(viejo, nuevo)
    if binario:
        open(ruta, 'wb').write(s)
    else:
        open(ruta, 'w', encoding='utf-8', newline='').write(s)
    print(nombre, 'ok', len(cambios))
