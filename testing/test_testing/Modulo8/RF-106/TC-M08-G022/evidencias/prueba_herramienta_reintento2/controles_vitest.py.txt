"""Controles por mutacion (Vitest, G022 reintento 2; copia del de reintento 1 con rutas del contenedor): mutan una COPIA de la vista en modo binario y la restauran."""
import json, os, subprocess, sys

FE = '/home/claude/g031r1/fe'
VISTA = 'src/prediction/pages/HistorialPredictivoView.tsx'
SPEC = 'testing/test_testing/Modulo8/RF-106/TC-M08-G022/tc-m08-g022.test.tsx'

CONTROLES = [
    ('tvco_solo_porcentaje', [(b"{datos.tvco.texto}</div>", b"{String(datos.tvco.porcentaje ?? '') + '%'}</div>")],
     ['CP-01', 'CP-03', 'CP-04', 'CP-05', 'CP-07', 'CP-08']),
    ('conflicto_sin_icono', [(b"{conflicto ? '\xe2\x9a\xa0' : ICONO_VALORACION", b"{false ? '\xe2\x9a\xa0' : ICONO_VALORACION")], ['CP-06']),
    ('pendientes_invertidos', [(b"{datos.pendientes.items.map((p) => (", b"{[...datos.pendientes.items].reverse().map((p) => (")], ['CP-09', 'CP-10']),
    ('siguiente_no_pagina', [(b"onClick={() => setPagina((x) => x + 1)}", b"onClick={() => setPagina((x) => x)}")], ['CP-10']),
]

salida = []
for nombre, reemplazos, esperadas in CONTROLES:
    ruta = os.path.join(FE, VISTA)
    original = open(ruta, 'rb').read()
    mut = original
    for viejo, nuevo in reemplazos:
        n = mut.count(viejo)
        assert n == 1, f'{nombre}: {n} apariciones de {viejo[:70]!r}'
        mut = mut.replace(viejo, nuevo)
    st = f'/home/claude/g022r2/ph/vitest_control_{nombre}'
    try:
        open(ruta, 'wb').write(mut)
        env = dict(os.environ, TZ='America/Bogota', STAGING_DIR=st, REINTENTO='2', RUN_ID=f'control-{nombre}')
        p = subprocess.run(['pnpm', 'exec', 'vitest', 'run', SPEC, '--reporter=verbose'], cwd=FE, env=env, capture_output=True, text=True, timeout=400)
        open(f'{st}_consola.txt', 'w').write(p.stdout + p.stderr)
    finally:
        open(ruta, 'wb').write(original)
    rep = json.load(open(f'{st}/parcial_vitest_TC-M08-G022_reintento2.json'))
    fallas = [c['paso'] for c in rep['checkpoints'] if c['estado'] == 'FALLA']
    no_alc = [c['paso'] for c in rep['checkpoints'] if 'No alcanzado' in c['obtenido']]
    prec = all(x['ok'] for x in rep['precondiciones'])
    ok = sorted(fallas) == sorted(esperadas) and not no_alc
    linea = {'control': nombre, 'exit': p.returncode, 'precondiciones_ok': prec, 'fallas': fallas, 'esperadas': esperadas,
             'no_alcanzado': no_alc, 'como_previsto': ok,
             'obtenidos_falla': {c['paso']: c['obtenido'][:220] for c in rep['checkpoints'] if c['estado'] == 'FALLA'}}
    salida.append(linea)
    print(json.dumps(linea, ensure_ascii=False))
json.dump(salida, open('/home/claude/g022r2/ph/vitest_controles_resumen.json', 'w'), ensure_ascii=False, indent=2)
sys.exit(0 if all(x['como_previsto'] for x in salida) else 1)
