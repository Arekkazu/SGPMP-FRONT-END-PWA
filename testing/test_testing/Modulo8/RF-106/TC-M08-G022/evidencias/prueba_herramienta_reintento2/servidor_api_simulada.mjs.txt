// Servidor de prueba de la herramienta (TC-M08-G022, parte Playwright). Sirve un build local en /sigab-test/ y una API
// simulada en /sigab-test/api con la forma real del backend. No es parte de la corrida oficial.
// Uso: node servidor_api_simulada.mjs.txt <dir_build> <escenario: normal|backend_tvco|backend_orden|backend_evaluables> <puerto>
import http from 'node:http';
import { readFileSync, existsSync, statSync, appendFileSync } from 'node:fs';
import { join, extname } from 'node:path';

const [dirBuild, escenario = 'normal', puertoTxt = '4173'] = process.argv.slice(2);
const PUERTO = Number(puertoTxt);
const LOG = process.env.MOCK_LOG ?? 'mock.log';
const HORA = 3_600_000;
const ms = (s) => new Date(s).getTime();
const iso = (s) => new Date(s).toISOString().replace('.000Z', '+00:00');

// 885 sintetico: 8 predicciones entre 2026-07-01 y 2026-10-10, 2 con valoracion y 6 sin valorar
const uuid = (n) => `dd00000${n}-0000-4000-8000-${String(n).padStart(12, '0')}`;
const preds = [
  ['2026-07-01T15:00:00Z', 1],
  ['2026-07-15T15:00:00Z', 2],
  ['2026-08-01T15:00:00Z', 3],
  ['2026-08-15T15:00:00Z', 0],
  ['2026-09-01T15:00:00Z', 2],
  ['2026-09-15T15:00:00Z', 1],
  ['2026-10-01T15:00:00Z', 3],
  ['2026-10-10T15:00:00Z', 0],
].map(([f, n], i) => ({
  clase: 'PREDICCION',
  fecha: f,
  datos: {
    id_resultado_inferencia: uuid(i + 1),
    nivel_riesgo: n,
    confianza: 0.85,
    condicion: `Condición QA ${i + 1}`,
    modo_sombra: false,
    tiene_retroalimentacion: i < 2,
    en_conflicto: false,
  },
}));

const retros = [
  { clase: 'RETROALIMENTACION', fecha: '2026-07-02T15:00:00Z', datos: { id_resultado_inferencia: uuid(1), valoracion: 'CORRECTO', veterinario: 'Vet QA', observaciones: null, en_conflicto: false } },
  { clase: 'RETROALIMENTACION', fecha: '2026-07-16T15:00:00Z', datos: { id_resultado_inferencia: uuid(2), valoracion: 'INCORRECTO', veterinario: 'Vet QA', observaciones: null, en_conflicto: false } },
];

const serial = (m) => ({ tipo: m.clase, fecha: iso(m.fecha), ...m.datos });

function calcularTvco() {
  const vals = new Map();
  for (const r of retros) {
    const id = r.datos.id_resultado_inferencia;
    if (!vals.has(id)) vals.set(id, new Set());
    vals.get(id).add(r.datos.valoracion);
  }
  let sinRetro = 0; let conflicto = 0; let validadas = 0; let correctas = 0;
  for (const p of preds) {
    const s = vals.get(p.datos.id_resultado_inferencia);
    if (!s || s.size === 0) sinRetro += 1;
    else if (s.size > 1) conflicto += 1;
    else { validadas += 1; if (s.has('CORRECTO')) correctas += 1; }
  }
  const porcentaje = Math.round((correctas / validadas) * 10000) / 100;
  const tasaError = Math.round((100 - porcentaje) * 100) / 100;
  const excluidas = sinRetro + conflicto;
  let texto = `${porcentaje}% (${validadas} ${validadas === 1 ? 'validada' : 'validadas'}`;
  if (excluidas > 0) {
    const partes = [];
    if (conflicto > 0) partes.push(sinRetro === 0 ? 'conflicto' : `${conflicto} conflicto`);
    if (sinRetro > 0) partes.push(conflicto === 0 ? 'sin retro' : `${sinRetro} sin retro`);
    texto += `; ${excluidas} ${excluidas === 1 ? 'excluida' : 'excluidas'}: ${partes.join(' + ')}`;
  }
  texto += ')';
  return { porcentaje, texto, tvco_denominador: validadas, tvco_excluidas_conflicto: conflicto, excluidas_sin_retroalimentacion: sinRetro, tasa_error: tasaError };
}

function historial(params) {
  const tvco = calcularTvco();
  if (escenario === 'backend_tvco') tvco.texto = tvco.texto.replace('50%', '49%');
  const conRetro = new Set(retros.map((r) => r.datos.id_resultado_inferencia));
  const ahora = Date.now();
  let pend = preds
    .filter((p) => !conRetro.has(p.datos.id_resultado_inferencia) && ahora - ms(p.fecha) > 48 * HORA)
    .sort((a, b) => (b.datos.nivel_riesgo - a.datos.nivel_riesgo) || (ms(b.fecha) - ms(a.fecha)));
  if (escenario === 'backend_orden') pend = [...pend].reverse();
  // Reintento 2 (control backend_evaluables): reproduce el filtro del backend que fallo en el reintento 1 (commit 655bc00af,
  // evaluables(): nivel >= 2 que cambia respecto de la anterior, o con alerta), para comprobar que CP-12 da FALLA.
  if (escenario === 'backend_evaluables') {
    const evaluables = new Set(); let anterior = null;
    for (const p of [...preds].sort((a, b) => ms(a.fecha) - ms(b.fecha))) {
      const n = p.datos.nivel_riesgo;
      if ((n !== null && n >= 2 && n !== anterior) || p.datos.tiene_alerta_generada) evaluables.add(p.datos.id_resultado_inferencia);
      anterior = n;
    }
    pend = pend.filter((p) => evaluables.has(p.datos.id_resultado_inferencia));
  }
  return {
    activo: { id_activo_biologico: 885, identificador: 'QA-G88-266-IND-001', especie: 'Qa', finca: 'Finca 184', estado: 'ACTIVO' },
    rango_fechas_aplicado: { fecha_inicio: params.get('fecha_inicio'), fecha_fin: params.get('fecha_fin') },
    ventana_correspondencia_horas: 12,
    predicciones: preds.map(serial),
    eventos_sanitarios: [],
    retroalimentaciones: retros.map(serial),
    grupos: [],
    correspondencias: [],
    tvco,
    pendientes: {
      total: pend.length,
      pagina: 1,
      por_pagina: 20,
      items: pend.map((p) => ({
        id_resultado_inferencia: p.datos.id_resultado_inferencia,
        fecha_inferencia: iso(p.fecha),
        nivel_riesgo: p.datos.nivel_riesgo,
        condicion: p.datos.condicion,
      })),
    },
    metadatos: {
      total_inferencias: preds.length,
      total_eventos_sanitarios: 0,
      total_retroalimentaciones: retros.length,
      total_pendientes: pend.length,
      total_conflictos: tvco.tvco_excluidas_conflicto,
    },
    fuentes_no_disponibles: [],
    mensaje: null,
    modo_visualizacion: params.get('modo_visualizacion') ?? 'LINEAL',
  };
}

const b64 = (o) => Buffer.from(JSON.stringify(o)).toString('base64url');
const TOKEN = `${b64({ alg: 'HS256', typ: 'JWT' })}.${b64({ sub: '97', jti: 'qa', rol: 3, exp: Math.floor(Date.now() / 1000) + 7200, iat: Math.floor(Date.now() / 1000) })}.firma`;
let conSesion = false;
const PERMISOS = [
  { id_recurso: 9106, nombre_recurso: 'historial_clinico_predictivo', id_accion: 2 },
  { id_recurso: 42, nombre_recurso: 'historial_diagnostico', id_accion: 2 },
  { id_recurso: 45, nombre_recurso: 'retroalimentacion_clinica', id_accion: 2 },
];
const TIPOS = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css', '.svg': 'image/svg+xml', '.png': 'image/png', '.json': 'application/json', '.woff2': 'font/woff2' };

http.createServer((req, res) => {
  const url = new URL(req.url, `http://127.0.0.1:${PUERTO}`);
  const json = (code, body) => { res.writeHead(code, { 'content-type': 'application/json' }); res.end(JSON.stringify(body)); };
  if (url.pathname.startsWith('/sigab-test/api/')) {
    const r = url.pathname.slice('/sigab-test/api'.length);
    appendFileSync(LOG, `${req.method} ${r}${url.search}\n`);
    if (req.method === 'POST' && /^\/sesiones\/?$/.test(r)) { conSesion = true; return json(200, { token: TOKEN, tipo: 'bearer', expira_en: 7200, message: 'ok' }); }
    if (req.method === 'POST' && r.startsWith('/sesiones/refresh')) return conSesion ? json(200, { token: TOKEN, tipo: 'bearer', expira_en: 7200, message: 'ok' }) : json(401, { error_code: 'SESION_INVALIDA', message: 'Sin sesión', fields: [] });
    if (req.method === 'DELETE' && /^\/sesiones\/?$/.test(r)) { conSesion = false; return json(200, { message: 'ok' }); }
    if (req.method !== 'GET') return json(201, { qa: 'escritura que la guardia debio abortar' });
    if (r === '/usuarios/me') return json(200, { id_usuario: 97, nombre: 'Vet', apellidos: 'QA', correo_electronico: 'vet@qa.test', nombre_rol: 'VETERINARIO', estado_cuenta: 'ACTIVA' });
    if (r.startsWith('/sesiones/me/permisos')) return json(200, { permisos: PERMISOS });
    if (r.startsWith('/reportes/opciones/activo')) return json(200, [{ valor: 885, etiqueta: 'QA-G88-266-IND-001 · Qa Semaforo Dinamico' }]);
    if (r.startsWith('/historial-predictivo/885')) return json(200, historial(url.searchParams));
    if (r.startsWith('/configuracion/mensajes-error')) return json(200, []);
    return json(404, { error_code: 'RUTA_NO_SIMULADA', message: `No simulada ${r}`, fields: [] });
  }
  let ruta = url.pathname.replace(/^\/sigab-test\/?/, '');
  let archivo = join(dirBuild, ruta || 'index.html');
  if (!existsSync(archivo) || statSync(archivo).isDirectory()) archivo = join(dirBuild, 'index.html');
  res.writeHead(200, { 'content-type': TIPOS[extname(archivo)] ?? 'application/octet-stream' });
  res.end(readFileSync(archivo));
}).listen(PUERTO, '127.0.0.1', () => console.log(`mock ${escenario} en ${PUERTO} sirviendo ${dirBuild}`));
