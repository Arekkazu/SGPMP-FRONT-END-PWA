type Clase = 'PREDICCION' | 'EVENTO_SANITARIO' | 'RETROALIMENTACION';
interface Fuente { clase: Clase; nombre: string; fecha: string; datos: Record<string, unknown> }

/** Fecha como datetime.isoformat() de Python (UTC, "+00:00"). */
const iso = (s: string) => s.replace('Z', '+00:00');
const ms = (s: string) => new Date(s).getTime();

function pred(idActivo: number, n: number, fecha: string, nivel: number): Fuente {
  const id = `c${String(idActivo).padStart(7, '0')}-0000-4000-8000-${String(n).padStart(12, '0')}`;
  return {
    clase: 'PREDICCION', nombre: `P${n}`, fecha,
    datos: {
      id_resultado_inferencia: id, nivel_riesgo: nivel, confianza: 0.9, condicion: `QA-${idActivo}-P${n}`,
      modo_sombra: false, tiene_retroalimentacion: false, en_conflicto: false,
    },
  };
}
function retro(idActivo: number, nPred: number, fecha: string, valoracion: string, veterinario: string): Fuente {
  const id = `c${String(idActivo).padStart(7, '0')}-0000-4000-8000-${String(nPred).padStart(12, '0')}`;
  return {
    clase: 'RETROALIMENTACION', nombre: `R${nPred}`, fecha,
    datos: {
      id_resultado_inferencia: id, valoracion, veterinario, observaciones: null, en_conflicto: false,
    },
  };
}
function serial(f: Fuente, extras?: Record<string, unknown>): Record<string, unknown> {
  return { tipo: f.clase, fecha: iso(f.fecha), ...f.datos, ...(extras ?? {}) };
}

// historial_predictivo.py:91-104
function correspondenciasBackend(preds: Fuente[], eventos: Fuente[], ventanaHoras: number) {
  const pares: Array<{ id_resultado_inferencia: unknown; id_evento: unknown; diferencia_minutos: number; diferencia: string; colision: boolean }> = [];
  for (const p of preds) {
    for (const e of eventos) {
      const delta = Math.abs(ms(e.fecha) - ms(p.fecha));
      if (delta <= ventanaHoras * HORA_MS) {
        const minutos = Math.floor(delta / 60_000);
        pares.push({
          id_resultado_inferencia: p.datos.id_resultado_inferencia, id_evento: e.datos.id_evento, diferencia_minutos: minutos,
          diferencia: `Diferencia: ${Math.floor(minutos / 60)} h${minutos % 60 ? ` ${minutos % 60} min` : ''}`, colision: delta <= HORA_MS,
        });
      }
    }
  }
  return pares;
}

// historial_predictivo.py:80-88 (y caso de uso :129: solo grupos de mas de 1)
function gruposBackend(marcadores: Fuente[]): Fuente[][] {
  const ordenados = [...marcadores].sort((a, b) => ms(a.fecha) - ms(b.fecha));
  const grupos: Fuente[][] = [];
  for (const m of ordenados) {
    const ultimo = grupos.length ? grupos[grupos.length - 1] : null;
    if (ultimo && ms(m.fecha) - ms(ultimo[ultimo.length - 1].fecha) <= HORA_MS) ultimo.push(m);
    else grupos.push([m]);
  }
  return grupos
    .map((g) => [...g].sort((a, b) => (PRIORIDAD[a.clase] - PRIORIDAD[b.clase]) || (ms(a.fecha) - ms(b.fecha))))
    .filter((g) => g.length > 1);
}

// historial_predictivo.py:52-70
function calcularTvco(preds: Fuente[], retros: Fuente[]) {
  const valoraciones = new Map<string, Set<string>>();
  for (const r of retros) {
    const id = String(r.datos.id_resultado_inferencia);
    if (!valoraciones.has(id)) valoraciones.set(id, new Set());
    valoraciones.get(id)!.add(String(r.datos.valoracion));
  }
  let sinRetro = 0;
  let conflicto = 0;
  let validadas = 0;
  let correctas = 0;
  for (const p of preds) {
    const id = String(p.datos.id_resultado_inferencia);
    const vals = valoraciones.get(id);
    if (!vals || vals.size === 0) sinRetro += 1;
    else if (vals.size > 1) conflicto += 1;
    else {
      validadas += 1;
      if (vals.has('CORRECTO')) correctas += 1;
    }
  }
  if (validadas === 0) {
    return {
      valoraciones,
      tvco: {
        porcentaje: null as number | null,
        texto: 'Sin datos suficientes para calcular',
        tvco_denominador: 0,
        tvco_excluidas_conflicto: conflicto,
        excluidas_sin_retroalimentacion: sinRetro,
        tasa_error: null as number | null,
      },
    };
  }
  const porcentaje = Math.round((correctas / validadas) * 10000) / 100;
  const tasaError = Math.round((100 - porcentaje) * 100) / 100;
  const excluidas = sinRetro + conflicto;
  let texto = `${porcentaje}% (${validadas} ${validadas === 1 ? 'validada' : 'validadas'}`;
  if (excluidas > 0) {
    const partes: string[] = [];
    if (conflicto > 0) partes.push(sinRetro === 0 ? 'conflicto' : `${conflicto} conflicto`);
    if (sinRetro > 0) partes.push(conflicto === 0 ? 'sin retro' : `${sinRetro} sin retro`);
    texto += `; ${excluidas} ${excluidas === 1 ? 'excluida' : 'excluidas'}: ${partes.join(' + ')}`;
  }
  texto += ')';
  return {
    valoraciones,
    tvco: {
      porcentaje,
      texto,
      tvco_denominador: validadas,
      tvco_excluidas_conflicto: conflicto,
      excluidas_sin_retroalimentacion: sinRetro,
      tasa_error: tasaError,
    },
  };
}

function construirRespuesta(idActivo: number, params: URLSearchParams) {
  const esc = ESCENARIOS[idActivo];
  const incluirEventos = params.get('incluir_eventos_sanitarios') !== 'false';
  const incluirRetro = params.get('incluir_retroalimentacion') !== 'false';
  const niveles = params.getAll('nivel_riesgo').map(Number);
  const pagina = Math.max(1, Number(params.get('pagina_pendientes') ?? '1') || 1);
  const porPagina = 20;

  const visibles = esc.preds.filter((p) => niveles.length === 0 || niveles.includes(Number(p.datos.nivel_riesgo)));
  const eventos = incluirEventos ? esc.eventos : [];
  const retros = incluirRetro ? esc.retros : [];
  const { valoraciones, tvco } = calcularTvco(esc.preds, esc.retros);
  const enConflicto = (id: unknown) => (valoraciones.get(String(id))?.size ?? 0) > 1;
  const tieneRetro = (id: unknown) => (valoraciones.get(String(id))?.size ?? 0) > 0;

  const ahora = ms(RELOJ);
  const pendientes = esc.preds
    .filter((p) => !tieneRetro(p.datos.id_resultado_inferencia) && ahora - ms(p.fecha) > 48 * HORA_MS)
    .sort((a, b) => (Number(b.datos.nivel_riesgo ?? 0) - Number(a.datos.nivel_riesgo ?? 0)) || (ms(b.fecha) - ms(a.fecha)));
  const inicio = (pagina - 1) * porPagina;
  const itemsPagina = pendientes.slice(inicio, inicio + porPagina);

  return {
    activo: { id_activo_biologico: idActivo, identificador: `QA-${idActivo}`, especie: 'Bovino', finca: 'Finca QA 184', estado: 'ACTIVO' },
    rango_fechas_aplicado: { fecha_inicio: params.get('fecha_inicio') ?? '2025-11-20', fecha_fin: params.get('fecha_fin') ?? '2026-11-20' },
    ventana_correspondencia_horas: VENTANA_SIMULADOR,
    predicciones: visibles.map((p) => serial(p, {
      tiene_retroalimentacion: tieneRetro(p.datos.id_resultado_inferencia),
      en_conflicto: enConflicto(p.datos.id_resultado_inferencia),
    })),
    eventos_sanitarios: eventos.map((e) => serial(e)),
    retroalimentaciones: retros.map((r) => serial(r, {
      en_conflicto: enConflicto(r.datos.id_resultado_inferencia),
    })),
    grupos: gruposBackend([...visibles, ...eventos, ...retros]).map((g) => g.map((m) => serial(m))),
    correspondencias: correspondenciasBackend(visibles, eventos, VENTANA_SIMULADOR),
    tvco,
    pendientes: {
      total: pendientes.length, pagina, por_pagina: porPagina,
      items: itemsPagina.map((p) => ({
        id_resultado_inferencia: p.datos.id_resultado_inferencia,
        fecha_inferencia: iso(p.fecha),
        nivel_riesgo: p.datos.nivel_riesgo,
        condicion: p.datos.condicion,
      })),
    },
    metadatos: {
      total_inferencias: esc.preds.length,
      total_eventos_sanitarios: esc.eventos.length,
      total_retroalimentaciones: esc.retros.length,
      total_pendientes: pendientes.length,
      total_conflictos: tvco.tvco_excluidas_conflicto,
    },
    fuentes_no_disponibles: [] as string[],
    mensaje: null as string | null,
    modo_visualizacion: params.get('modo_visualizacion') ?? 'LINEAL',
  };
}

function adaptadorSimulado(config: InternalAxiosRequestConfig): Promise<AxiosResponse> {
  const metodo = (config.method ?? 'get').toUpperCase();
  const uri = http.getUri(config);
  const url = config.url ?? '';
  peticiones.push({ metodo, url, uri });
  const ok = (data: unknown): Promise<AxiosResponse> => Promise.resolve({ data, status: 200, statusText: 'OK', headers: {}, config, request: {} });
  if (metodo === 'GET' && url.includes('/reportes/opciones/activo')) {
    return ok(Object.entries(ESCENARIOS).map(([valor, e]) => ({ valor: Number(valor), etiqueta: e.etiqueta })));
  }
  const m = url.match(/\/historial-predictivo\/(\d+)/);
  if (metodo === 'GET' && m && ESCENARIOS[Number(m[1])]) {
    const params = new URLSearchParams(uri.includes('?') ? uri.slice(uri.indexOf('?') + 1) : '');
    const data = construirRespuesta(Number(m[1]), params); if (Number(m[1]) === 5006) { data.retroalimentaciones.forEach((r) => { r.en_conflicto = false; }); data.predicciones.forEach((p) => { p.en_conflicto = false; }); data.metadatos.total_conflictos = 0; }
    ultimaRespuesta = data;
    return ok(data);
  }
  const body = { error_code: 'RUTA_NO_SIMULADA', message: `Ruta no simulada: ${metodo} ${url}`, fields: [], timestamp: RELOJ };
  const resp: AxiosResponse = { data: body, status: 404, statusText: 'Not Found', headers: {}, config, request: {} };
  return Promise.reject(new AxiosError('Request failed with status code 404', 'ERR_BAD_REQUEST', config, {}, resp));
}

const norm = (s: string | null | undefined) => (s ?? '').replace(/[\u00A0\u202F]/g, ' ').replace(/\s+/g, ' ').trim();
const ICONOS_CONFLICTO = new Set(['⚠', '⚠️', '▲', '△']);

function tvcoTexto(): string {
  const titulo = Array.from(document.querySelectorAll('div, span, p, h1, h2, h3, h4, dt, label'))
    .find((el) => el.children.length === 0 && norm(el.textContent) === 'Tasa de validación clínica');
  return titulo?.nextElementSibling ? norm(titulo.nextElementSibling.textContent) : '(sin TVCO)';
}
function textoVisible(): string {
  return norm(document.body.textContent);
}
function iconosRetro(): string[] {
  return Array.from(document.querySelectorAll('.recharts-scatter-symbol text')).map((t) => norm(t.textContent));
}
function itemsPendientes(): Array<{ texto: string; botones: string[] }> {
  const h3 = Array.from(document.querySelectorAll('h3')).find((el) => norm(el.textContent) === 'Predicciones sin evaluar (más de 48 horas)');
  const cont = h3?.parentElement;
  if (!cont) return [];
  return Array.from(cont.querySelectorAll('ul > li')).map((li) => ({
    texto: norm(li.textContent),
    botones: Array.from(li.querySelectorAll('button')).map((b) => norm(b.textContent)),
  }));
}
function tasaVisible(n: number): { ok: boolean; detalle: string } {
  const txt = textoVisible();
  const re = new RegExp(`error[^%\\d]{0,40}(?<![\\d.,])${n}([.,]0+)? ?%|(?<![\\d.,])${n}([.,]0+)? ?%[^\\d]{0,40}error`, 'i');
  const m = txt.match(re);
  if (m) return { ok: true, detalle: m[0] };
  return { ok: false, detalle: `(sin tasa de error visible; TVCO="${tvcoTexto()}")` };
}

function firma(): string {
  return [
    document.querySelectorAll('.recharts-scatter-symbol').length,
    document.querySelectorAll('li').length,
    tvcoTexto(),
    screen.queryByText('No se pudo cargar el historial') !== null,
  ].join('|');
}
async function esperarEstable(): Promise<void> {
  await waitFor(() => expect(screen.queryByText('Cargando historial...')).toBeNull(), { timeout: 3000 });
  let anterior = '';
  for (let i = 0; i < 60; i += 1) {
    await new Promise((r) => setTimeout(r, 50));
    const actual = firma();
    if (actual === anterior) return;
    anterior = actual;
  }
}
async function seleccionar(etiqueta: string): Promise<string> {
  try {
    const combo = await screen.findByRole('combobox', { name: /Animal o lote/ });
    await waitFor(() => expect(peticiones.some((p) => p.url.includes('/reportes/opciones/activo'))).toBe(true));
    await new Promise((r) => setTimeout(r, 0));
    const antes = peticiones.length;
    fireEvent.change(combo, { target: { value: etiqueta } });
    await waitFor(() => expect(peticiones.slice(antes).some((p) => /\/historial-predictivo\/\d+/.test(p.url))).toBe(true), { timeout: 3000 });
    await esperarEstable();
    return '';
  } catch (e) {
    return `no se pudo (${String((e as Error)?.message ?? e).split('\n')[0].slice(0, 160)})`;
  }
}
