let logins = 0;
async function guardia(route: Route, req: Request): Promise<void> {
  const metodo = req.method();
  if (metodo === 'GET' || metodo === 'HEAD' || metodo === 'OPTIONS') return route.continue();
  const url = new URL(req.url());
  const ruta = url.pathname;
  if (/googleapis\.com|gstatic\.com|firebaseio\.com|firebase\.com/.test(url.hostname)) {
    escrituras.push({ metodo, ruta: `${url.hostname}${ruta}`, resultado: 'abortada (host externo)' });
    return route.abort();
  }
  if (metodo === 'POST' && /\/sesiones\/?$/.test(ruta)) {
    logins += 1;
    escrituras.push({ metodo, ruta, resultado: `al servidor (login ${logins})` });
    if (logins > 1) { violaciones.push(`segundo login (${ruta})`); return route.abort(); }
    return route.continue();
  }
  if ((metodo === 'POST' && /\/sesiones\/refresh\/?$/.test(ruta)) || (metodo === 'DELETE' && /\/sesiones\/?$/.test(ruta))) {
    escrituras.push({ metodo, ruta, resultado: 'al servidor (sesion)' });
    return route.continue();
  }
  if (metodo === 'POST' && /\/usuarios\/me\/fcm-token\/?$/.test(ruta)) {
    escrituras.push({ metodo, ruta, resultado: 'abortada (registro de push)' });
    return route.abort();
  }
  escrituras.push({ metodo, ruta, resultado: 'ABORTADA (no permitida)' });
  violaciones.push(`${metodo} ${ruta}`);
  return route.abort();
}

const postsSimulados: Array<Record<string, unknown>> = [];
async function simularRegistro(route: Route, req: Request): Promise<void> {
  if (req.method() !== 'POST') return route.fallback();
  const body = (req.postDataJSON() ?? {}) as Record<string, unknown>;
  postsSimulados.push(body);
  const ruta = new URL(req.url()).pathname;
  escrituras.push({ metodo: 'POST', ruta, resultado: 'simulada con page.route (no llega a TEST)' });
  return route.fulfill({
    status: 201,
    contentType: 'application/json',
    body: JSON.stringify({
      id_retroalimentacion: '0a023000-0000-4000-8000-000000000001',
      id_resultado_inferencia: body.id_resultado_inferencia,
      id_activo_biologico: body.id_activo_biologico,
      estado_retroalimentacion: body.estado_retroalimentacion,
      diagnosticos_reales: body.diagnosticos_reales ?? null,
      fuente_diagnostico: body.fuente_diagnostico ?? null,
      es_fuente_desconocida: body.fuente_diagnostico == null,
      es_conflicto_retroalimentacion: false,
      observaciones_clinicas: body.observaciones_clinicas ?? null,
      id_usuario_veterinario: 97,
      fecha_retroalimentacion: new Date().toISOString(),
      estado_registro: 'ACTIVO',
    }),
  });
}

const norm = (s: string | null | undefined) => (s ?? '').replace(/[\u00A0\u202F]/g, ' ').replace(/\s+/g, ' ').trim();

}

async function esperarEstable(page: Page): Promise<void> {
  await expect(page.getByText('Cargando historial...')).toHaveCount(0, { timeout: 30_000 });
  let anterior = '';
  const limite = Date.now() + 10_000;
  while (Date.now() < limite) {
    await page.waitForTimeout(250);
    const f = await page.evaluate(() => [
      document.querySelectorAll('.recharts-scatter-symbol').length,
      document.querySelectorAll('li').length,
    ].join('|'));
    if (f === anterior) return;
    anterior = f;
  }
}

async function elegirActivo(page: Page, etiqueta: string): Promise<Response> {
  const esHistorial = (r: Response) => new RegExp(`/historial-predictivo/${ACTIVO}\\?`).test(r.url())
    && r.url().includes('modo_visualizacion=LINEAL')
    && r.request().method() === 'GET';
  const respLineal = page.waitForResponse(esHistorial, { timeout: 30_000 });
  const combo = page.getByRole('combobox', { name: 'Animal o lote' });
  await combo.fill(etiqueta);
  const opcionLista = page.getByRole('option', { name: etiqueta, exact: true });
  if (await opcionLista.count()) await opcionLista.first().click();
  return respLineal;
}

function esNoGetApi(p: RegistroPeticion): boolean {
  if (p.metodo === 'GET' || p.metodo === 'HEAD' || p.metodo === 'OPTIONS') return false;
  if (/googleapis\.com|gstatic\.com|firebaseio\.com|firebase\.com/.test(p.host)) return false;
  if (/\/sesiones(\/refresh)?\/?$/.test(p.ruta)) return false;
  if (/\/usuarios\/me\/fcm-token\/?$/.test(p.ruta)) return false;
  return true;
}

      .filter((p) => p.t <= tCambioUrl && p.pantalla.endsWith('/prediccion/historial-predictivo') && esNoGetApi(p));

    let infoBandeja = 'sin respuesta';
    if (respBandeja) {
      let bodyB: BandejaResp | null = null;
      try { bodyB = (await respBandeja.json()) as BandejaResp; } catch { bodyB = null; }
      const hallado = bodyB?.items?.find((it) => it.id_resultado_inferencia === item.id_resultado_inferencia);
      infoBandeja = `HTTP=${respBandeja.status()}, x-request-id="${respBandeja.headers()['x-request-id'] ?? ''}", total=${bodyB?.total ?? '?'}, ventana_dias=${bodyB?.ventana_dias ?? '?'}, id_en_items=${hallado ? `sí (estado_evaluacion=${hallado.estado_evaluacion}, puede_evaluar=${hallado.puede_evaluar})` : 'no'}`;
    }

    const dialogo = page.getByRole('dialog', { name: 'Retroalimentación clínica', exact: true });
    const limite = Date.now() + 8_000;
    let dialogoVisible = false;
    while (Date.now() < limite) {
      if (await dialogo.isVisible().catch(() => false)) { dialogoVisible = true; break; }
      await page.waitForTimeout(200);
    }

    const uActual = new URL(page.url());
    const rutaYSearch = `${uActual.pathname}${uActual.search}`;
    let titleContexto = '';
    let lineaActivo = '';
    let textoDialogo = '';
    if (dialogoVisible) {
      const titles = await dialogo.locator('[title]').evaluateAll((els) => els.map((e) => e.getAttribute('title') ?? ''));
      titleContexto = titles.find((t) => t === item.id_resultado_inferencia) ?? titles[0] ?? '';
      textoDialogo = norm(await dialogo.textContent());
      const inner = await dialogo.innerText().catch(() => '');
      lineaActivo = norm(inner.split('\n').find((l) => l.includes('Activo biológico:')) ?? '');
    }

    const ok = uActual.pathname.endsWith('/prediccion/retroalimentacion')
      && uActual.searchParams.get('resultado') === item.id_resultado_inferencia
      && uActual.searchParams.get('activo') === String(ACTIVO)
      && dialogoVisible
      && titleContexto === item.id_resultado_inferencia
      && textoDialogo.includes(identificadorEsperado);

    const obtenido = `url="${rutaYSearch}" | bandeja={${infoBandeja}} | dialogo=${dialogoVisible ? 'sí' : 'no'}${dialogoVisible ? ` (title="${titleContexto}", linea_activo="${lineaActivo}")` : ''}`;
    await captura(page, nombreCaptura);
    return { ok, obtenido, dialogoVisible, noGetEnClic };
  }

  // -- 4. CP-01: pendiente de nivel >= 2
  const res01 = await probarClicPendiente(idxAlto, itemAlto!, 'CP-01_formulario_rf72');
  anotar('CP-01', res01.ok, res01.obtenido);

  // -- 5. CP-03: registro CORRECTO (POST simulado)
  if (!res01.dialogoVisible) {
    anotar('CP-03', false, 'formulario no abierto en CP-01: no se pudo registrar');
  } else {
    const dialogo = page.getByRole('dialog', { name: 'Retroalimentación clínica', exact: true });
    await dialogo.getByRole('radio', { name: /^Correcto/ }).click();
    await dialogo.getByRole('button', { name: 'Registrar evaluación' }).click();

    const limPost = Date.now() + 15_000;
    while (Date.now() < limPost && postsSimulados.length < 1) {
      await page.waitForTimeout(200);
    }
    const limUi = Date.now() + 10_000;
    let dialogoCerrado = false;
    let avisoVisible = false;
    while (Date.now() < limUi) {
      dialogoCerrado = !(await dialogo.isVisible().catch(() => false));
      avisoVisible = await page.getByText('Evaluación registrada').isVisible().catch(() => false);
      if (dialogoCerrado && avisoVisible) break;
      await page.waitForTimeout(200);
    }
    const nPosts = postsSimulados.length;
    const b0 = postsSimulados[0] ?? null;
    const ok03 = nPosts === 1
      && b0?.id_resultado_inferencia === itemAlto!.id_resultado_inferencia
      && Number(b0?.id_activo_biologico) === ACTIVO
      && b0?.estado_retroalimentacion === 'CORRECTO'
      && dialogoCerrado
      && avisoVisible;
    anotar(
      'CP-03',
      ok03,
      `posts=${nPosts}, body=${b0 ? `{id_resultado_inferencia="${b0.id_resultado_inferencia}", id_activo_biologico=${b0.id_activo_biologico}, estado_retroalimentacion="${b0.estado_retroalimentacion}"}` : 'null'}, dialogo_cerrado=${dialogoCerrado}, aviso=${avisoVisible}`,
    );
    const uPost = new URL(page.url());
    observaciones.push(`Ruta tras el registro en RF-72: ${uPost.pathname}${uPost.search}.`);
  }

  // -- 6. Vuelta al historial y CP-06
  const getsHistAntesVuelta = peticiones.filter((p) => p.metodo === 'GET' && new RegExp(`/historial-predictivo/${ACTIVO}$`).test(p.ruta)).length;
  await page.goBack();
  await page.waitForURL((u) => u.pathname.endsWith('/prediccion/historial-predictivo'), { timeout: 15_000 }).catch(() => undefined);
  await page.waitForTimeout(3_000);
  const valorComboAlVolver = await page.getByRole('combobox', { name: 'Animal o lote' }).inputValue().catch(() => '(no disponible)');
  const getsAutoAlVolver = peticiones.filter((p) => p.metodo === 'GET' && /\/historial-predictivo\//.test(p.ruta)).length - getsHistAntesVuelta;
  observaciones.push(`Al volver con atrás: combobox "Animal o lote"="${valorComboAlVolver}", GET /historial-predictivo sin volver a elegir=${getsAutoAlVolver}.`);

  const r2 = await elegirActivo(page, opcion!.etiqueta);
  await esperarEstable(page);
  const h2 = (await r2.json()) as Historial;

  const getsHistDespuesVuelta = peticiones.filter((p) => p.metodo === 'GET' && new RegExp(`/historial-predictivo/${ACTIVO}$`).test(p.ruta)).length;
  const tvcoCardLoc = page.locator('xpath=//div[normalize-space(.)="Tasa de validación clínica"]/following-sibling::div[1]');
  const tvcoPantalla2 = (await tvcoCardLoc.count()) ? norm(await tvcoCardLoc.first().textContent()) : '(sin TVCO)';
  const lisDom2 = await page.evaluate(() => {
    const n = (s: string | null | undefined) => (s ?? '').replace(/[\u00A0\u202F]/g, ' ').replace(/\s+/g, ' ').trim();
    const h3 = Array.from(document.querySelectorAll('h3')).find((el) => n(el.textContent) === 'Predicciones sin evaluar (más de 48 horas)');
    const cont = h3?.parentElement;
    if (!cont) return [] as Array<{ texto: string; botones: string[] }>;
    return Array.from(cont.querySelectorAll('ul > li')).map((li) => ({
      texto: n(li.textContent),
      botones: Array.from(li.querySelectorAll('button')).map((b) => n(b.textContent)),
    }));
  });
  const domOk2 = lisDom2.length === h2.pendientes.items.length && h2.pendientes.items.every((it, i) => {
    const fecha = fechaBogota(it.fecha_inferencia);
    const nivel = NIVEL_LABEL[Number(it.nivel_riesgo ?? 0)] ?? '';
    return lisDom2[i].texto.includes(fecha) && lisDom2[i].texto.includes(nivel) && lisDom2[i].botones.includes('Registrar retroalimentación');
  });
  const idAltoSigue = h2.pendientes.items.some((p) => p.id_resultado_inferencia === itemAlto!.id_resultado_inferencia);
  const ok06 = getsHistDespuesVuelta > getsHistAntesVuelta && r2.status() === 200 && tvcoPantalla2 === h2.tvco.texto && domOk2;
  anotar(
    'CP-06',
    ok06,
    `HTTP=${r2.status()}, x-request-id="${r2.headers()['x-request-id'] ?? ''}", respuesta_tvco="${h2.tvco.texto}", pantalla_tvco="${tvcoPantalla2}", li=${lisDom2.length}/${h2.pendientes.items.length} (domOk=${domOk2}), id_CP01_sigue_en_pendientes=${idAltoSigue}`,
  );
  observaciones.push(`Al volver al historial, el id de CP-01 (${corto(itemAlto!.id_resultado_inferencia)}) ${idAltoSigue ? 'sigue' : 'no sigue'} en pendientes (con POST simulado se espera que siga).`);

  // -- 7. CP-02: pendiente de nivel <= 1
  const idxBajo2 = h2.pendientes.items.findIndex((p) => p.id_resultado_inferencia === itemBajo!.id_resultado_inferencia);
  precondicion(
    'El pendiente de nivel <= 1 de la línea base sigue en la lista',
    idxBajo2 >= 0,
    idxBajo2 >= 0 ? `indice=${idxBajo2}, id=${corto(itemBajo!.id_resultado_inferencia)}` : `id=${corto(itemBajo!.id_resultado_inferencia)} ausente`,
  );
  const res02 = await probarClicPendiente(idxBajo2, itemBajo!, 'CP-02_rf72');
  anotar('CP-02', res02.ok, res02.obtenido);

  // -- 8. CP-04: fuentes que consulta la vista
  const enHistorial = peticiones.filter((p) => p.pantalla.endsWith('/prediccion/historial-predictivo'));
  const getsHist = enHistorial.filter((p) => p.metodo === 'GET' && new RegExp(`/historial-predictivo/${ACTIVO}$`).test(p.ruta));
  const infsHist = enHistorial.filter((p) => p.ruta.includes('/prediccion/inferencias'));
  const infsOtras = peticiones.filter((p) => !p.pantalla.endsWith('/prediccion/historial-predictivo') && p.ruta.includes('/prediccion/inferencias'));
  const ok04 = getsHist.length >= 2 && getsHist.every((p) => p.status === 200) && infsHist.length === 0;
  anotar(
    'CP-04',
    ok04,
    `GET /historial-predictivo/${ACTIVO}=${getsHist.length} (status=[${getsHist.map((p) => p.status ?? 'sin_resp').join(', ')}]); /prediccion/inferencias en historial=${infsHist.length}${infsHist.length ? ` [${infsHist.map((p) => `${p.metodo} ${p.ruta}`).join(', ')}]` : ''}`,
  );
  observaciones.push(`Peticiones a /prediccion/inferencias en otras pantallas: ${infsOtras.length}. Respuestas clave registradas: ${respuestasClave.length}.`);

  // -- 9. CP-05: la vista es de solo lectura
  const noGetHistorial = enHistorial.filter(esNoGetApi);
  const ok05 = estatica.dialogs === 0
    && estatica.radiogroups === 0
    && estatica.textareas === 0
    && estatica.botonesValoracion.length === 0
    && botonesPendNoPermitidos.length === 0
    && noGetHistorial.length === 0
    && res01.noGetEnClic.length === 0
    && res02.noGetEnClic.length === 0;
  anotar(
    'CP-05',
    ok05,
    `dialogs=${estatica.dialogs}, radiogroups=${estatica.radiogroups}, textareas=${estatica.textareas}, botones_valoracion=${estatica.botonesValoracion.length} [${estatica.botonesValoracion.join(', ')}], botones_pend=[${estatica.botonesPend.join(', ')}], no_get_historial=${noGetHistorial.length}, no_get_clic_cp01=${res01.noGetEnClic.length}, no_get_clic_cp02=${res02.noGetEnClic.length}`,
  );

  // -- 10. Precondicion de despliegue con los scripts que cargo la app
  const todo = scripts.map((s) => s.cuerpo).join('\n');
  const index = [...new Set(scripts.map((s) => s.nombre).filter((n) => /^index-.*\.js$/.test(n)))];
  await page.goto('about:blank');
  observaciones.push(`Despliegue: bundle ${index.join(', ') || '(no identificado)'} en ${BASE}.`);
  precondicion(
    'El frontend desplegado contiene los textos y ruta de RF-106 / RF-72',
    todo.includes('Registrar retroalimentación')
      && todo.includes('Predicciones sin evaluar (más de 48 horas)')
      && todo.includes('Registrar evaluación')
      && todo.includes('/prediccion/retroalimentacion?resultado='),
    `bundle: ${index.join(', ') || '(sin index-*.js)'}; ${scripts.length} scripts revisados`,
  );

  // -- 11. Verificacion final
  expect(violaciones, violaciones.join('; ')).toEqual([]);
  expect(CPS.every((c) => c.estado === 'OK'), CPS.filter((c) => c.estado === 'FALLA').map((c) => c.paso).join(', ')).toBe(true);
});
