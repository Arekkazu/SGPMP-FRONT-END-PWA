let histCalls = 0;

function calcularTvco(preds, retros) {
  const vals = new Map();
  for (const r of retros) {
    if (!vals.has(r.id_resultado_inferencia)) vals.set(r.id_resultado_inferencia, new Set());
    vals.get(r.id_resultado_inferencia).add(r.valoracion);
  }
  let sinRetro = 0; let conflicto = 0; let validadas = 0; let correctas = 0;
  for (const p of preds) {
    const s = vals.get(p.id_resultado_inferencia);
    if (!s || s.size === 0) sinRetro += 1;
    else if (s.size > 1) conflicto += 1;
    else { validadas += 1; if (s.has('CORRECTO')) correctas += 1; }
  }
  const porcentaje = Math.round((correctas / validadas) * 10000) / 100;
  const tasaError = Math.round((100 - porcentaje) * 100) / 100;
  const excluidas = sinRetro + conflicto;
  let texto = `${porcentaje}% (${validadas} ${validadas === 1 ? 'validada' : 'validadas'}`;
  if (excluidas > 0) {
    const partes = [];
    if (conflicto > 0) partes.push(sinRetro === 0 ? 'conflicto' : `${conflicto} conflicto`);
    if (sinRetro > 0) partes.push(conflicto === 0 ? 'sin retro' : `${sinRetro} sin retro`);
    texto += `; ${excluidas} ${excluidas === 1 ? 'excluida' : 'excluidas'}: ${partes.join(' + ')}`;
  }
  texto += ')';
  return { porcentaje, texto, tvco_denominador: validadas, tvco_excluidas_conflicto: conflicto, excluidas_sin_retroalimentacion: sinRetro, tasa_error: tasaError };
}

function historial(params) {
  histCalls += 1;
  const cambiarP1 = escenario === 'cambia_al_volver' && histCalls >= 2;
  const retros = cambiarP1
    ? [
      ...baseRetros,
      {
        tipo: 'RETROALIMENTACION',
        fecha: iso(AHORA - 1000),
        id_resultado_inferencia: uuid(1),
        valoracion: 'CORRECTO',
        veterinario: 'QA Vet',
        observaciones: null,
        en_conflicto: false,
      },
    ]
    : [...baseRetros];

  const conRetro = new Set(retros.map((r) => r.id_resultado_inferencia));
  const preds = basePreds.map((p) => ({
    tipo: 'PREDICCION',
    fecha: iso(p.fecha),
    id_resultado_inferencia: p.id,
    nivel_riesgo: p.nivel,
    confianza: 0.85,
    condicion: p.condicion,
    modo_sombra: false,
    tiene_retroalimentacion: conRetro.has(p.id),
    en_conflicto: false,
  }));

  const tvco = calcularTvco(preds, retros);
  const pend = basePreds
    .filter((p) => !conRetro.has(p.id) && AHORA - p.fecha > 48 * HORA)
    .sort((a, b) => (b.nivel - a.nivel) || (b.fecha - a.fecha));

  return {
    activo: { id_activo_biologico: 885, identificador: 'QA-G88-266-IND-001', especie: 'Qa', finca: 'Finca 184', estado: 'ACTIVO' },
    rango_fechas_aplicado: { fecha_inicio: params.get('fecha_inicio'), fecha_fin: params.get('fecha_fin') },
    ventana_correspondencia_horas: 12,
    predicciones: preds,
    eventos_sanitarios: [],
    retroalimentaciones: retros,
    grupos: [],
    correspondencias: [],
    tvco,
    pendientes: {
      total: pend.length,
      pagina: 1,
      por_pagina: 20,
      items: pend.map((p) => ({
        id_resultado_inferencia: p.id,
        fecha_inferencia: iso(p.fecha),
        nivel_riesgo: p.nivel,
        condicion: p.condicion,
      })),
    },
    metadatos: {
      total_inferencias: preds.length,
      total_eventos_sanitarios: 0,
      total_retroalimentaciones: retros.length,
      total_pendientes: pend.length,
      total_conflictos: tvco.tvco_excluidas_conflicto,
    },
    fuentes_no_disponibles: [],
    mensaje: null,
    modo_visualizacion: params.get('modo_visualizacion') ?? 'LINEAL',
  };
}

function bandeja(params) {
  const idActivo = Number(params.get('id_activo_biologico'));
  const items = [];
  if (idActivo === 885) {
    if (escenario !== 'bandeja_sin_alto') {
      items.push(itemBandeja(basePreds[0], { estado_evaluacion: 'PENDIENTE', puede_evaluar: true, evaluada_por_mi: false, numero_evaluaciones: 0 }));
    }
    if (escenario === 'bandeja_con_bajo') {
      items.push(itemBandeja(basePreds[2], { estado_evaluacion: 'PENDIENTE', puede_evaluar: true, evaluada_por_mi: false, numero_evaluaciones: 0 }));
    }
    items.push(itemBandeja(basePreds[4], { estado_evaluacion: 'EVALUADO', puede_evaluar: false, evaluada_por_mi: true, numero_evaluaciones: 1 }));
  }
  const pendientes = items.filter((i) => i.estado_evaluacion === 'PENDIENTE').length;
  const evaluadas = items.filter((i) => i.estado_evaluacion === 'EVALUADO').length;
  const vencidas = items.filter((i) => i.estado_evaluacion === 'VENTANA_VENCIDA').length;
  return {
    items,
    total: items.length,
    pagina: 1,
    por_pagina: 25,
    ventana_dias: 90,
    indicadores: { pendientes, evaluadas, vencidas },
  };
}

let conSesion = false;
const PERMISOS = [
  { id_recurso: 9106, nombre_recurso: 'historial_clinico_predictivo', id_accion: 2 },
  { id_recurso: 42, nombre_recurso: 'historial_diagnostico', id_accion: 2 },

const TIPOS = { '.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css', '.svg': 'image/svg+xml', '.png': 'image/png', '.json': 'application/json', '.woff2': 'font/woff2' };

http.createServer((req, res) => {
  const url = new URL(req.url, `http://127.0.0.1:${PUERTO}`);
  const rid = `qa-g023-${++reqSeq}`;
  const json = (code, body) => { res.writeHead(code, { 'content-type': 'application/json', 'x-request-id': rid }); res.end(JSON.stringify(body)); };
  if (url.pathname.startsWith('/sigab-test/api/')) {
    const r = url.pathname.slice('/sigab-test/api'.length);
    if (req.method === 'POST' && /^\/sesiones\/?$/.test(r)) {
      appendFileSync(LOG, `${req.method} ${r}${url.search}\n`);
      conSesion = true;
      return json(200, { token: TOKEN, tipo: 'bearer', expira_en: 7200, message: 'ok' });
    }
    if (req.method === 'POST' && r.startsWith('/sesiones/refresh')) {
      appendFileSync(LOG, `${req.method} ${r}${url.search}\n`);
      return conSesion ? json(200, { token: TOKEN, tipo: 'bearer', expira_en: 7200, message: 'ok' }) : json(401, { error_code: 'SESION_INVALIDA', message: 'Sin sesión', fields: [] });
    }
    if (req.method === 'DELETE' && /^\/sesiones\/?$/.test(r)) {
      appendFileSync(LOG, `${req.method} ${r}${url.search}\n`);
      conSesion = false;
      return json(200, { message: 'ok' });
    }
    if (req.method !== 'GET') {
      appendFileSync(LOG, `ESCRITURA_NO_PERMITIDA ${req.method} ${r}${url.search}\n`);
      return json(201, { qa: 'escritura que la guardia o page.route debio atender' });
    }
    appendFileSync(LOG, `${req.method} ${r}${url.search}\n`);
    if (r === '/usuarios/me') return json(200, { id_usuario: 97, nombre: 'Vet', apellidos: 'QA', correo_electronico: 'vet@qa.test', nombre_rol: 'VETERINARIO', estado_cuenta: 'ACTIVA' });
    if (r.startsWith('/sesiones/me/permisos')) return json(200, { permisos: PERMISOS });
    if (r.startsWith('/reportes/opciones/activo')) return json(200, [{ valor: 885, etiqueta: 'QA-G88-266-IND-001 · Qa Semaforo Dinamico' }]);
    if (r.startsWith('/historial-predictivo/885')) return json(200, historial(url.searchParams));
    if (r.startsWith('/prediccion/retroalimentacion/bandeja')) return json(200, bandeja(url.searchParams));
    if (r.startsWith('/prediccion/patologias')) return json(200, { total: 0, items: [] });
    if (r.startsWith('/prediccion/inferencias')) return json(200, { total: 0, items: [] });
    if (r.startsWith('/configuracion/mensajes-error')) return json(200, []);
    return json(404, { error_code: 'RUTA_NO_SIMULADA', message: `No simulada ${r}`, fields: [] });
  }
  let ruta = url.pathname.replace(/^\/sigab-test\/?/, '');
  let archivo = join(dirBuild, ruta || 'index.html');
  if (!existsSync(archivo) || statSync(archivo).isDirectory()) archivo = join(dirBuild, 'index.html');
  res.writeHead(200, { 'content-type': TIPOS[extname(archivo)] ?? 'application/octet-stream' });
  res.end(readFileSync(archivo));
}).listen(PUERTO, '127.0.0.1', () => console.log(`mock ${escenario} en ${PUERTO} sirviendo ${dirBuild}`));
