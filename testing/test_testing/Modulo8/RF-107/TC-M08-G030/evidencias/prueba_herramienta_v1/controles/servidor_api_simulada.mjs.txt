// Base64url con espacio inicial antes de '{' para evitar prefijo JWT en el payload
// manteniendo compatibilidad con JSON.parse(atob(...)) de AuthContext.
const b64ConEspacio = (o) => Buffer.from(' ' + JSON.stringify(o)).toString('base64url');
let reqSeq = 0;
let tokenSeq = 0;
let sesionSeq = 0;

/** @type {Map<string, { cuenta: typeof CUENTA_VET, tokenActivo: string | null }>} */
const sesiones = new Map();
/** @type {Map<string, { sesionId: string, cuenta: typeof CUENTA_VET, revocado: boolean }>} */
const tokens = new Map();
let ultimaSesionId = null;

function emitirToken(sesionId, cuenta) {
  const ses = sesiones.get(sesionId);
  if (ses && ses.tokenActivo) {
    const prev = tokens.get(ses.tokenActivo);
    if (prev) prev.revocado = true;
  }
  const jti = `qa-g030-${++tokenSeq}`;
  const ahoraSeg = Math.floor(Date.now() / 1000);
  const tok = `token-mock.${b64ConEspacio({ sub: String(cuenta.id), jti, rol: cuenta.rol, exp: ahoraSeg + 7200, iat: ahoraSeg })}.firma`;
  tokens.set(tok, { sesionId, cuenta, revocado: false });
  if (ses) ses.tokenActivo = tok;
  return tok;
}

function leerCookieSesion(req) {
  const cookie = req.headers.cookie || '';
  const m = /(?:^|;\s*)qa_sid=([^;]+)/.exec(cookie);
  return m ? m[1] : null;
}

function leerBody(req) {
  return new Promise((resolve) => {
    const chunks = [];
    req.on('data', (c) => chunks.push(c));
    req.on('end', () => resolve(Buffer.concat(chunks).toString('utf8')));
  });
}

function resolverCuentaDesdeAuth(req) {
  const auth = req.headers.authorization || '';
  const m = /^Bearer\s+(.+)$/i.exec(auth);
  if (!m) {
    const sid = leerCookieSesion(req) || ultimaSesionId;
    const ses = sid ? sesiones.get(sid) : null;
    return { ok: true, cuenta: ses ? ses.cuenta : CUENTA_VET };
  }
  const info = tokens.get(m[1]);
  if (!info) return { ok: false, code: 401, error_code: 'TOKEN_INVALIDO' };
  if (info.revocado) return { ok: false, code: 401, error_code: 'TOKEN_REVOCADO' };
  return { ok: true, cuenta: info.cuenta };
}

const TIPOS = {
  '.html': 'text/html',
  '.js': 'text/javascript',
  '.css': 'text/css',
  '.svg': 'image/svg+xml',
  '.png': 'image/png',
  '.json': 'application/json',
  '.woff2': 'font/woff2',
};

http.createServer(async (req, res) => {
  const url = new URL(req.url, `http://127.0.0.1:${PUERTO}`);
  const rid = `qa-g030-${++reqSeq}`;
  const json = (code, body, extraHeaders = {}) => {
    res.writeHead(code, { 'content-type': 'application/json', 'x-request-id': rid, ...extraHeaders });
    res.end(JSON.stringify(body));
  };

  if (url.pathname.startsWith('/sigab-test/api/')) {
    const r = url.pathname.slice('/sigab-test/api'.length);

    if (req.method === 'POST' && /^\/sesiones\/?$/.test(r)) {
      const raw = await leerBody(req);
      let email = '';
      try {
        const parsed = JSON.parse(raw || '{}');
        email = String(parsed.correo_electronico || parsed.email || '').toLowerCase();
      } catch { /* ignore */ }
      const cuenta = email.includes('admin') ? CUENTA_ADMIN : CUENTA_VET;
      const sid = `sid-${++sesionSeq}-${cuenta.clave}`;
      sesiones.set(sid, { cuenta, tokenActivo: null });
      ultimaSesionId = sid;
      const tok = emitirToken(sid, cuenta);
      appendFileSync(LOG, `${req.method} ${r}${url.search} cuenta=${cuenta.clave}\n`);
      return json(200, { token: tok, tipo: 'bearer', expira_en: 7200, message: 'ok' }, {
        'set-cookie': `qa_sid=${sid}; Path=/; HttpOnly; SameSite=Lax`,
      });
    }

    if (req.method === 'POST' && r.startsWith('/sesiones/refresh')) {
      const sid = leerCookieSesion(req);
      const ses = sid ? sesiones.get(sid) : null;
      appendFileSync(LOG, `${req.method} ${r}${url.search} sid=${sid || 'none'}\n`);
      if (!ses) return json(401, { error_code: 'SESION_INVALIDA', message: 'Sin sesión', fields: [] });
      const tok = emitirToken(sid, ses.cuenta);
      return json(200, { token: tok, tipo: 'bearer', expira_en: 7200, message: 'ok' });
    }

    if (req.method === 'DELETE' && /^\/sesiones\/?$/.test(r)) {
      const sid = leerCookieSesion(req);
      const ses = sid ? sesiones.get(sid) : null;
      if (ses && ses.tokenActivo) {
        const info = tokens.get(ses.tokenActivo);
        if (info) info.revocado = true;
      }
      if (sid) sesiones.delete(sid);
      appendFileSync(LOG, `${req.method} ${r}${url.search} sid=${sid || 'none'}\n`);
      return json(200, { message: 'ok' }, {
        'set-cookie': 'qa_sid=; Path=/; Max-Age=0',
      });
    }

    if (req.method !== 'GET') {
      appendFileSync(LOG, `ESCRITURA_NO_PERMITIDA ${req.method} ${r}${url.search}\n`);
      return json(201, { qa: 'escritura que la guardia debio abortar' });
    }

    const auth = resolverCuentaDesdeAuth(req);
    if (!auth.ok) {
      appendFileSync(LOG, `${req.method} ${r}${url.search} -> 401 ${auth.error_code}\n`);
      return json(401, { error_code: auth.error_code, message: 'Token revocado o invalido', fields: [] });
    }
    const cuenta = auth.cuenta;
    appendFileSync(LOG, `${req.method} ${r}${url.search} cuenta=${cuenta.clave}\n`);

    if (r === '/usuarios/me') return json(200, cuenta.usuario);
    if (r === '/usuarios/me/fincas') return json(200, [{ id_finca: 184, nombre_finca: 'Finca 184' }]);
    if (r.startsWith('/sesiones/me/permisos')) return json(200, { permisos: cuenta.permisos });
    if (r.startsWith('/configuracion/mensajes-error')) return json(200, []);
    if (r.startsWith('/configuracion/interfaz/contexto')) return json(200, {});
    if (r.startsWith('/configuracion/personalizacion/idioma')) return json(200, { idioma: 'es-CO' });
    if (r.startsWith('/configuracion/personalizacion/tema')) return json(200, { tema: 'light' });
    if (r.startsWith('/configuracion/personalizacion/dashboard/widgets')) return json(200, []);
    if (r.startsWith('/configuracion/personalizacion/dashboard')) return json(200, {});
    if (r.startsWith('/notificaciones')) return json(200, { items: [], total: 0, no_leidas: 0 });

    if (r.startsWith('/reportes/opciones/activo')) return json(200, [{ valor: 885, etiqueta: 'QA-G88-266-IND-001 · Qa Semaforo Dinamico' }]);
    if (r.startsWith('/historial-predictivo/885')) return json(200, historial(url.searchParams));

    if (r === '/activos-biologicos' || r === '/activos-biologicos/') {
      return json(200, { items: [ACTIVO_885], total: 1, pagina: 1, por_pagina: 20 });
    }
    if (r === '/activos-biologicos/885/ficha-integral') return json(200, FICHA_885);
    if (r === '/activos-biologicos/885/eventos') return json(200, []);
    if (r === '/activos-biologicos/885') return json(200, ACTIVO_885);

    if (r.startsWith('/finanzas/titulares')) return json(200, TITULARES);
    if (r === '/finanzas/periodos' || r === '/finanzas/periodos/') return json(200, PERIODOS);
    if (r === '/finanzas/periodos/101/prevalidacion') return json(200, PREVALIDACION_101);
    if (r === '/finanzas/periodos/101/comprobantes') return json(200, []);
    if (r.startsWith('/finanzas/reconocimientos/propuestas')) return json(200, []);

    return json(404, { error_code: 'RUTA_NO_SIMULADA', message: `No simulada ${r}`, fields: [] });
  }

  let ruta = url.pathname.replace(/^\/sigab-test\/?/, '');
  let archivo = join(dirBuild, ruta || 'index.html');
  if (!existsSync(archivo) || statSync(archivo).isDirectory()) archivo = join(dirBuild, 'index.html');
  res.writeHead(200, { 'content-type': TIPOS[extname(archivo)] ?? 'application/octet-stream' });
  res.end(readFileSync(archivo));
}).listen(PUERTO, '127.0.0.1', () => console.log(`mock ${escenario} en ${PUERTO} sirviendo ${dirBuild}`));
